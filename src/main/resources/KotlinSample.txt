    /**
     * Maps known numeric sample codes to labels and returns other for unmatched values.
     *
     * @param code integer operand or sample code
     *
     * @return selected sample label
     *
     * @author Mohammad ali Riazati
     * @since 1.0.0
     */
    fun selectLabel(code: Int): String {

        return when (code) {

            1    -> "first"

            22   -> "second"

            333  -> "third"

            4444 -> "fourth"

            else -> "other"

        }
    }

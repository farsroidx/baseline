/**
 * Returns the prefix followed by the top-level sample identifier.
 *
 * @param prefix string segment concatenated in the supplied order
 *
 * @return concatenated sample text
 *
 * @author Mohammad ali Riazati
 * @since 1.0.0
 */
fun describeSample(prefix: String): String {

    return prefix + id

}

/**
 * Returns the sum of two integers while exercising local declaration spacing.
 *
 * @param first integer operand or sample code
 * @param second integer operand or sample code
 *
 * @return sum of the supplied integers
 *
 * @author Mohammad ali Riazati
 * @since 1.0.0
 */
fun calculateSample(first: Int, second: Int): Int {

    val total = first + second;

    return total

}

/**
 * Concatenates the three supplied strings in order.
 *
 * @param first string segment concatenated in the supplied order
 * @param second string segment concatenated in the supplied order
 * @param third string segment concatenated in the supplied order
 *
 * @return concatenated sample text
 *
 * @author Mohammad ali Riazati
 * @since 1.0.0
 */
fun combineSample(first: String, second: String, third: String): String {

    return first + second + third

}

/**
 * Returns the sum of the four supplied integers.
 *
 * @param first integer operand or sample code
 * @param second integer operand or sample code
 * @param third integer operand or sample code
 * @param fourth integer operand or sample code
 *
 * @return sum of the supplied integers
 *
 * @author Mohammad ali Riazati
 * @since 1.0.0
 */
fun totalSample(first: Int, second: Int, third: Int, fourth: Int): Int {

    return first + second + third + fourth

}

/**
 * Concatenates five descriptions in their supplied order.

 * @param extremelyLongFirstDescription string segment concatenated in the supplied order
 * @param extremelyLongSecondDescription string segment concatenated in the supplied order
 * @param extremelyLongThirdDescription string segment concatenated in the supplied order

 * @param extremelyLongFifthDescription string segment concatenated in the supplied order

    extremelyLongFirstDescription: String,
    extremelyLongSecondDescription: String,
    extremelyLongThirdDescription: String,
    extremelyLongFourthDescription: String,

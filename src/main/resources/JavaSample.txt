    /**
     * Returns a label for known sample codes, or "other" for an unmatched code.
     *
     * @param code numeric sample code to map to a label
     * @return selected sample label
     */
    public String selectLabel(int code) {

        switch (code) {

            case 1    :

            return "first";

            case 22   :

            return "second";

            case 333  :

            return "third";

            case 4444 :

            return "fourth";

            default   :

            return "other";

        }
    }

    /**
     * Returns the prefix followed by the instance identifier.
     *
     * @param prefix first string segment
     * @return concatenated string
     */
    public String describe(String prefix) {

        String description = prefix + id;

        return description;

    }

    /**
     * Returns the sum of the two integers.
     *
     * @param first  integer operand added to the result
     * @param second integer operand added to the result
     * @return sum of the operands
     */
    public int calculate(int first, int second) {

        int sum = first + second;

        return sum;

    }

    /**
     * Concatenates three strings in their supplied order.
     *
     * @param first  string segment concatenated in the supplied order
     * @param second string segment concatenated in the supplied order
     * @param third  string segment concatenated in the supplied order
     * @return concatenated string
     */
    public String combine(String first, String second, String third) {

        return first + second + third;

    }

    /**
     * Returns the sum of four integers.
     *
     * @param first  integer operand added to the result
     * @param second integer operand added to the result
     * @param third  integer operand added to the result
     * @param fourth integer operand added to the result
     * @return sum of the operands
     */
    public int total(int first, int second, int third, int fourth) {

        return first + second + third + fourth;

    }

    /**
     * Concatenates five description segments and writes a diagnostic message to standard output.
     *
     * @param extremelyLongFirstDescription  string segment concatenated in the supplied order
     * @param extremelyLongSecondDescription string segment concatenated in the supplied order
     * @param extremelyLongThirdDescription  string segment concatenated in the supplied order
     * @param extremelyLongFourthDescription string segment concatenated in the supplied order
     * @param extremelyLongFifthDescription  string segment concatenated in the supplied order
     * @return concatenated string
     */
    public String buildDescription(
        String extremelyLongFirstDescription,
        String extremelyLongSecondDescription,
        String extremelyLongThirdDescription,
        String extremelyLongFourthDescription,
        String extremelyLongFifthDescription
    ) {

        String result = extremelyLongFirstDescription + extremelyLongSecondDescription;

        System.out.println( "An unrelated diagnostic message." );

        return result + extremelyLongThirdDescription + extremelyLongFourthDescription + extremelyLongFifthDescription;

    }
